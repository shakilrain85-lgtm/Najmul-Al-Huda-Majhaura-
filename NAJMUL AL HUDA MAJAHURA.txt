   REPORT SYSTEM
========================= */

let totalUsers =
parseInt(
    localStorage.getItem("najmTotalUsers") || "0"
);

let totalSearches =
parseInt(
    localStorage.getItem("najmTotalSearches") || "0"
);

let visitHistory =
JSON.parse(
    localStorage.getItem("najmVisitHistory") || "[]"
);


/* USER VISIT */

const now =
Date.now();

visitHistory.push(now);

visitHistory =
visitHistory.filter(
    time =>
    now - time <=
    60 * 60 * 1000
);

totalUsers++;

localStorage.setItem(
    "najmTotalUsers",
    totalUsers
);

localStorage.setItem(
    "najmVisitHistory",
    JSON.stringify(visitHistory)
);


/* SHOW REPORT */

function updateReport(){

    document.getElementById(
        "totalUsers"
    ).textContent =
        totalUsers;

    document.getElementById(
        "totalSearches"
    ).textContent =
        totalSearches;

    const currentTime =
        Date.now();

    const lastHour =
        visitHistory.filter(
            time =>
            currentTime - time <=
            60 * 60 * 1000
        );

    document.getElementById(
        "oneHourUsers"
    ).textContent =
        lastHour.length;
}

updateReport();

setInterval(
    updateReport,
    10000
);


/* =========================
   NAME SEARCH
========================= */

function searchName(){

    const input =
        document.getElementById(
            "nameInput"
        );

    const result =
        document.getElementById(
            "result"
        );

    const resultName =
        document.getElementById(
            "resultName"
        );

    const name =
        input.value.trim();


    if(name===""){

        input.focus();

        return;
    }


    totalSearches++;

    localStorage.setItem(
        "najmTotalSearches",
        totalSearches
    );


    resultName.textContent =
        "🌟 " + name;


    result.style.display =
        "block";


    result.scrollIntoView({

        behavior:"smooth",

        block:"center"

    });


    updateReport();

    createNameCelebration();
}

            searchName();
